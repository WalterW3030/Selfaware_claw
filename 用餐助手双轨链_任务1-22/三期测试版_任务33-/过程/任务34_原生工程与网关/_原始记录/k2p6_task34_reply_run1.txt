// 食知 AI 网关：extend.AI 通道（MODEL_MAP 定稿见第一部分总结报告§四）
// 密钥/环境全部走云开发环境，代码内不出现任何 key
const cloud = require('wx-server-sdk')

cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV })

const ai = cloud.ai() // wx-server-sdk >= 4.0.1

// —— 六功能位定稿映射（改这里即全链路生效）——
// 依据：第一部分总结报告§四「最终模型确认」
const MODEL_MAP = {
  recognize:  { channel: 'extend', model: 'deepseek-flash', thinking: 'off'    }, // 识图：原生多模态，思考off
  copy:       { channel: 'extend', model: 'hy3'                                }, // 文案：纯文本生成
  nutrition:  { channel: 'extend', model: 'hy3'                                }, // 营养：只翻译不出数字
  order_fast: { channel: 'extend', model: 'deepseek-flash', thinking: 'off'    }, // 点餐快速：思考off
  order_deep: { channel: 'extend', model: 'deepseek-flash', thinking: 'medium' }, // 点餐深度：思考medium（可上探high）
  chat:       { channel: 'extend', model: 'deepseek-flash', thinking: 'low'    }, // 自由提问：思考low
}

// thinking 档位 → extend.AI thinking 字段映射
// off → disabled；其余档位（low/medium/high）→ enabled + effort
function buildThinking(thinking) {
  if (!thinking) return null
  if (thinking === 'off') return { type: 'disabled' }
  return { type: 'enabled', effort: thinking }
}

// —— extend.AI 通道调用 ——
// 姿势依据：背景调研_模型与平台.md §1.3
// createModel('cloudbase') → generateText；图片以 base64 进 messages
async function callExtend(cfg, messages) {
  const model = ai.createModel('cloudbase')
  const data = { model: cfg.model, messages }
  const thinking = buildThinking(cfg.thinking)
  if (thinking) data.thinking = thinking
  return model.generateText(data)
}

// —— 从不同返回形态中取纯文本（nutrition 红线检查用）——
function extractText(res) {
  if (!res) return ''
  // generateText 常见返回形态兼容
  if (typeof res === 'string') return res
  if (typeof res.text === 'string') return res.text
  const choice = res.choices && res.choices[0]
  if (choice && choice.message && typeof choice.message.content === 'string') {
    return choice.message.content
  }
  if (res.result && typeof res.result === 'string') return res.result
  return ''
}

// —— nutrition 数字泄漏红线 ——
// 依据：工程总包 T2 红线，营养数值必须来自数据库，模型不得输出数字
const NUMERIC_LEAK_RE = /\d+(\.\d+)?\s*(kcal|千卡|卡路里|克|g|mg|毫克|微克|ug|μg)/i
const BARE_NUMERIC_RE = /\d+(\.\d+)?/

function guardNutrition(res, allowNumbers) {
  if (allowNumbers) return res
  const text = extractText(res)
  if (NUMERIC_LEAK_RE.test(text) || BARE_NUMERIC_RE.test(text)) {
    return {
      error: 'numeric_leak_blocked',
      note: '营养数值必须来自数据库，模型输出的数字被红线拦截',
    }
  }
  return res
}

exports.main = async (event) => {
  const { mode, messages } = event || {}

  const cfg = MODEL_MAP[mode]
  if (!cfg) {
    return { error: 'unknown_mode', mode }
  }
  if (!Array.isArray(messages) || messages.length === 0) {
    return { error: 'empty_messages' }
  }

  try {
    const res = await callExtend(cfg, messages)

    // nutrition 模式：返回前做数字泄漏检查
    if (mode === 'nutrition') {
      return guardNutrition(res, event.allowNumbers)
    }
    return res
  } catch (err) {
    return {
      error: 'extend_call_failed',
      mode,
      message: err && err.message ? err.message : String(err),
    }
  }
}

## `cloudfunctions/aiGateway/index.js`（并入 selfhost 后的完整版）

```javascript
// 食知 AI 网关：extend.AI 主通道 + selfhost 备选通道（默认关闭，仅当环境变量存在才启用）
// MODEL_MAP 定稿见第一部分总结报告§四；selfhost 通道为 2.3 增补
const cloud = require('wx-server-sdk')

cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV })

const ai = cloud.ai() // wx-server-sdk >= 4.0.1

// —— 六功能位定稿映射 ——
// channel 恒为 'extend'（主通道优先）；fallback 为可选自备通道模型，缺省即无备选
const MODEL_MAP = {
  recognize:  { channel: 'extend', model: 'deepseek-flash', thinking: 'off',    fallback: { model: 'qwen3-vl-flash' } },
  copy:       { channel: 'extend', model: 'hy3',                                fallback: { model: 'deepseek-flash' } },
  nutrition:  { channel: 'extend', model: 'hy3'                                                        },
  order_fast: { channel: 'extend', model: 'deepseek-flash', thinking: 'off',    fallback: { model: 'qwen3-vl-flash' } },
  order_deep: { channel: 'extend', model: 'deepseek-flash', thinking: 'medium', fallback: { model: 'qwen3-vl-235b' } },
  chat:       { channel: 'extend', model: 'deepseek-flash', thinking: 'low'                            },
}

// —— selfhost 通道是否可用：两个环境变量都存在才算启用（默认关闭）——
function selfhostReady() {
  return !!(process.env.SELFHOST_BASE_URL && process.env.SELFHOST_API_KEY)
}

// thinking 档位 → extend.AI thinking 字段映射
function buildThinking(thinking) {
  if (!thinking) return null
  if (thinking === 'off') return { type: 'disabled' }
  return { type: 'enabled', effort: thinking }
}

// —— extend.AI 通道 ——
async function callExtend(cfg, messages) {
  const model = ai.createModel('cloudbase')
  const data = { model: cfg.model, messages }
  const thinking = buildThinking(cfg.thinking)
  if (thinking) data.thinking = thinking
  return model.generateText(data)
}

// —— selfhost 备选通道：OpenAI 兼容端点（百炼/硅基流动等）——
// 仅在 selfhostReady() 为真时被调用；key 只在环境变量，绝不入代码
async function callSelfhost(modelName, messages) {
  const base = process.env.SELFHOST_BASE_URL.replace(/\/+$/, '')
  const resp = await fetch(base + '/chat/completions', {
    method: 'POST',
    headers: {
      Authorization: 'Bearer ' + process.env.SELFHOST_API_KEY,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify({ model: modelName, messages, stream: false }),
  })
  if (!resp.ok) {
    const detail = await resp.text().catch(() => '')
    throw new Error('selfhost_http_' + resp.status + ' ' + detail.slice(0, 200))
  }
  return resp.json()
}

// —— 提取纯文本（nutrition 红线检查用）——
function extractText(res) {
  if (!res) return ''
  if (typeof res === 'string') return res
  if (typeof res.text === 'string') return res.text
  const choice = res.choices && res.choices[0]
  if (choice && choice.message && typeof choice.message.content === 'string') {
    return choice.message.content
  }
  if (res.result && typeof res.result === 'string') return res.result
  return ''
}

// —— nutrition 数字泄漏红线 ——
const NUMERIC_LEAK_RE = /\d+(\.\d+)?\s*(kcal|千卡|卡路里|克|g|mg|毫克|微克|ug|μg)/i
const BARE_NUMERIC_RE = /\d+(\.\d+)?/

function guardNutrition(res, allowNumbers) {
  if (allowNumbers) return res
  const text = extractText(res)
  if (NUMERIC_LEAK_RE.test(text) || BARE_NUMERIC_RE.test(text)) {
    return {
      error: 'numeric_leak_blocked',
      note: '营养数值必须来自数据库，模型输出的数字被红线拦截',
    }
  }
  return res
}

exports.main = async (event) => {
  const { mode, messages, forceChannel } = event || {}

  const cfg = MODEL_MAP[mode]
  if (!cfg) return { error: 'unknown_mode', mode }
  if (!Array.isArray(messages) || messages.length === 0) return { error: 'empty_messages' }

  // —— 通道选择：extend 优先 ——
  // forceChannel==='selfhost' 且环境变量齐备时强制走备选（仅供联调对照）；
  // 否则恒走 extend；extend 失败且备选可用时自动退到 selfhost。
  const wantSelfhost = forceChannel === 'selfhost' && cfg.fallback && selfhostReady()
  let res
  let channelUsed = 'extend'
  let extendError = null

  if (wantSelfhost) {
    try {
      res = await callSelfhost(cfg.fallback.model, messages)
      channelUsed = 'selfhost'
    } catch (err) {
      extendError = err && err.message ? err.message : String(err)
      res = { error: 'selfhost_call_failed', message: extendError }
    }
  } else {
    try {
      res = await callExtend(cfg, messages)
    } catch (err) {
      extendError = err && err.message ? err.message : String(err)
      // extend 失败 → 备选可用则兜底
      if (cfg.fallback && selfhostReady()) {
        try {
          res = await callSelfhost(cfg.fallback.model, messages)
          channelUsed = 'selfhost'
        } catch (err2) {
          return {
            error: 'all_channels_failed',
            mode,
            extend: extendError,
            selfhost: err2 && err2.message ? err2.message : String(err2),
          }
        }
      } else {
        return { error: 'extend_call_failed', mode, message: extendError }
      }
    }
  }

  // nutrition：无论走哪条通道，返回前都做红线检查
  if (mode === 'nutrition') {
    return guardNutrition(res, event.allowNumbers)
  }
  return { channel: channelUsed, result: res }
}
```

关键行为说明：

- **默认关闭**：`selfhostReady()` 要求 `SELFHOST_BASE_URL` 与 `SELFHOST_API_KEY` 同时存在，缺一即视为未启用，主路径永远走 extend。
- **extend 优先**：无 `forceChannel` 时恒先调 extend；只有 extend 抛错且备选齐备才自动兜底。
- **仅供对照**：`forceChannel:'selfhost'` 是联调对照开关（对应 T2-H4），生产调用不会传此字段。
- **红线不因通道改变**：nutrition 在两条通道合并后统一过 `guardNutrition`。
- **返回结构**：主路径加了 `{ channel, result }` 包裹以便联调时确认实际通道；如前端 `gateway.js` 期望扁平结构，可按需在网关侧改为直接 `return res`（一行改动）。

| 本地调试步骤 + 测试 event | 文档§三、§四（`mode:"chat"` 示例） |
| 常见问题报错对照 | 文档§五 含环境 ID 未配 / 基库过低 / 模型未开通三类 |

## 遗留与提示

- `MODEL_MAP` 中 `fallback` 的模型名（`qwen3-vl-flash`/`qwen3-vl-235b`）为占位候选，实际取值以 Walter 在百炼/硅基流动控制台开通的模型 ID 为准，接入时改此一处即可。
- 返回结构本轮采用 `{ channel, result }` 包裹；若前端 `gateway.js`（2.1 轮原生模块）按扁平结构解包，两处需对齐——建议在 2.4 联调前统一，避免解包错位。
