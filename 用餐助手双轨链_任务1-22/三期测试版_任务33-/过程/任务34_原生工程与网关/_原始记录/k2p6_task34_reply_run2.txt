**`cloudfunctions/aiGateway/index.js`(第3轮定稿版)**

```javascript
// 食知AI网关:extend.AI 为主通道,selfhost 为备选(默认关闭)
// 依据:第一部分总结报告§四(六功能位定稿)+ 背景调研_模型与平台
const cloud = require('wx-server-sdk')
cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV })

// 需 wx-server-sdk >= 4.0.1 才提供 cloud.ai()
const ai = cloud.ai()

// ===== 六功能位模型定稿(改这里即全链路生效)=====
// channel: 'extend' = extend.AI平台模型(首选); 'selfhost' = 自备OpenAI兼容端点(备选)
const MODEL_MAP = {
  recognize:  { channel: 'extend', model: 'deepseek-flash', thinking: 'off' },
  copy:       { channel: 'extend', model: 'hy3' },
  nutrition:  { channel: 'extend', model: 'hy3' },                          // 只翻译不出数字
  order_fast: { channel: 'extend', model: 'deepseek-flash', thinking: 'off' },
  order_deep: { channel: 'extend', model: 'deepseek-flash', thinking: 'medium' },
  chat:       { channel: 'extend', model: 'deepseek-flash', thinking: 'low' },
}

// selfhost 通道备选模型名(仅当 channel 被改为 'selfhost' 时生效)
const SELFHOST_MODEL_FALLBACK = {
  recognize:  'qwen3-vl-flash',
  order_fast: 'qwen3-vl-flash',
  order_deep: 'Qwen3-VL-235B',
}

// thinking档位映射:off -> disabled;low/medium/high -> enabled + effort
function buildThinking(cfg) {
  if (!cfg.thinking) return null
  if (cfg.thinking === 'off') return { type: 'disabled' }
  return { type: 'enabled', effort: cfg.thinking }
}

// 统一提取返回文本(兼容不同返回形态)
function extractText(res) {
  if (!res) return ''
  if (typeof res.text === 'string') return res.text
  if (res.choices && res.choices[0] && res.choices[0].message) {
    return res.choices[0].message.content || ''
  }
  return ''
}

// selfhost 通道是否可用:两个环境变量都齐备才算开启(默认关闭)
function selfhostReady() {
  return !!(process.env.SELFHOST_BASE_URL && process.env.SELFHOST_API_KEY)
}

// ---- extend.AI 通道(主)----
// 图片消息形态(与官方教程一致,由调用方构造):
//   { role:'user', content:[
//       { type:'text', text:'...' },
//       { type:'image_url', image_url:{ url:'data:image/jpeg;base64,...' } }
//   ]}
async function callExtend(cfg, messages) {
  const model = ai.createModel('cloudbase')
  const data = { model: cfg.model, messages }
  const thinking = buildThinking(cfg)
  if (thinking) data.thinking = thinking
  return model.generateText(data)
}

// ---- selfhost 备选通道(OpenAI兼容端点;默认关闭)----
async function callSelfhost(cfg, messages) {
  if (!selfhostReady()) {
    throw new Error('selfhost_channel_not_configured')
  }
  const modelName = cfg.model || SELFHOST_MODEL_FALLBACK[cfg._mode]
  const resp = await fetch(process.env.SELFHOST_BASE_URL.replace(/\/$/, '') + '/chat/completions', {
    method: 'POST',
    headers: {
      'Authorization': 'Bearer ' + process.env.SELFHOST_API_KEY,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify({ model: modelName, messages, stream: false }),
  })
  return resp.json()
}

// 通道调度:extend 优先;当且仅当 cfg.channel==='selfhost' 且环境变量齐备才走 selfhost,
// 否则回落 extend,保证主链路不被备选通道缺失拖垮
async function dispatch(mode, cfg, messages) {
  if (cfg.channel === 'selfhost') {
    if (selfhostReady()) return callSelfhost({ ...cfg, _mode: mode }, messages)
    // 备选未配置 -> 回落主通道
    return callExtend(cfg, messages)
  }
  return callExtend(cfg, messages)
}

// ---- nutrition 数字泄漏红线:数值必须来自数据库,模型只做翻译 ----
const NUMERIC_LEAK = /\d+(\.\d+)?\s*(kcal|kj|千卡|大卡|千焦|克|毫克|g|mg)/i

exports.main = async (event) => {
  const { mode, messages } = event || {}
  const cfg = MODEL_MAP[mode]
  if (!cfg) return { error: 'unknown_mode', mode }

  let res
  try {
    res = await dispatch(mode, cfg, messages)
  } catch (e) {
    return { error: 'gateway_call_failed', detail: e.message }
  }

  // 营养模式返回前拦截数字泄漏
  if (mode === 'nutrition' && !event.allowNumbers) {
    const text = extractText(res)
    if (NUMERIC_LEAK.test(text)) {
      return {
        error: 'numeric_leak_blocked',
        note: '营养数值必须来自数据库,模型输出数字被拦截',
      }
    }
  }

  return res
}
```

第3轮相对第2轮的增量点:
- 新增 `selfhostReady()`:仅当 `SELFHOST_BASE_URL` 与 `SELFHOST_API_KEY` 同时存在才认为通道开启(默认关闭)。
- 新增 `dispatch()`:通道调度器,extend 优先;`channel==='selfhost'` 但环境变量缺失时**自动回落 extend**,不抛错、不断链。
- 新增 `SELFHOST_MODEL_FALLBACK`:selfhost 通道下各功能位的 VL 模型名(recognize/order_fast=qwen3-vl-flash,order_deep=Qwen3-VL-235B)。
- 主链路错误统一收敛为 `gateway_call_failed`。

`config.json` 与 `package.json` 同第2轮,无变化。
